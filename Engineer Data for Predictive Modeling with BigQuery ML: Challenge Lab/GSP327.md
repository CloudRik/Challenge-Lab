## ⚡ STEP-BY-STEP SOLUTION

## TASK-1 PROMPT:

```bash
CREATE OR REPLACE TABLE `<Enter_project_id>.taxirides.<Enter_Taxi_training_Data_name>` AS
SELECT
  (tolls_amount + fare_amount) AS <Enter_Fare_amount>,
  pickup_datetime,
  pickup_longitude,
  pickup_latitude,
  dropoff_longitude,
  dropoff_latitude,
  passenger_count,
  trip_distance
FROM
  `<Enter_project_id>.taxirides.historical_taxi_rides_raw`
WHERE
  trip_distance > 1
  AND fare_amount >= 3
  AND passenger_count > 1
  AND pickup_longitude BETWEEN -75 AND -73
  AND pickup_latitude BETWEEN 40 AND 42
  AND dropoff_longitude BETWEEN -75 AND -73
  AND dropoff_latitude BETWEEN 40 AND 42
  AND RAND() < 0.001
LIMIT 1000000;


```

<br>

## TASK-2 PROMPT:

> 1st Command:

```bash

CREATE OR REPLACE MODEL `<Enter_project_id>.taxirides.<fare_model_Name>`
TRANSFORM(
  <fare_amount_name>,
  pickup_datetime,
  passenger_count,
  trip_distance,
  pickup_longitude,
  pickup_latitude,
  dropoff_longitude,
  dropoff_latitude,
  ST_Distance(
    ST_GeogPoint(pickup_longitude, pickup_latitude),
    ST_GeogPoint(dropoff_longitude, dropoff_latitude)
  ) AS euclidean
)
OPTIONS(
  MODEL_TYPE='linear_reg',
  INPUT_LABEL_COLS=['<fare_amount_Name>']
)
AS
SELECT
  <fare_amount_Name>,
  pickup_datetime,
  passenger_count,
  trip_distance,
  pickup_longitude,
  pickup_latitude,
  dropoff_longitude,
  dropoff_latitude
FROM
  `<Project_id>.taxirides.<taxi_training_data_Name>`
WHERE
  <fare_amount_name> IS NOT NULL
  AND pickup_longitude IS NOT NULL
  AND pickup_latitude IS NOT NULL
  AND dropoff_longitude IS NOT NULL
  AND dropoff_latitude IS NOT NULL;


```


> 2st Command:

```

SELECT
  *
FROM
  ML.EVALUATE(MODEL `<PROJECT_ID>.taxirides.<fare_model_Name>`)
```

## TASK-3 PROMPT:
> 1st Command:

```bash
CREATE OR REPLACE MODEL `[YOUR_PROJECT_ID].taxirides.[YOUR_MODEL_NAME]`
TRANSFORM(
  [YOUR_FARE_COLUMN],
  pickup_datetime,
  passenger_count AS passengers,
  pickup_longitude AS pickuplon,
  pickup_latitude AS pickuplat,
  dropoff_longitude AS dropofflon,
  dropoff_latitude AS dropofflat,
  ST_Distance(
    ST_GeogPoint(pickup_longitude, pickup_latitude),
    ST_GeogPoint(dropoff_longitude, dropoff_latitude)
  ) AS euclidean
)
OPTIONS(
  MODEL_TYPE='linear_reg',
  INPUT_LABEL_COLS=['[YOUR_FARE_COLUMN]']
)
AS
SELECT
  (tolls_amount + fare_amount) AS [YOUR_FARE_COLUMN],
  pickup_datetime,
  passenger_count,
  pickup_longitude,
  pickup_latitude,
  dropoff_longitude,
  dropoff_latitude
FROM
  `[YOUR_PROJECT_ID].taxirides.historical_taxi_rides_raw`
WHERE
  trip_distance > 1
  AND fare_amount >= 3
  AND passenger_count > 1
  AND pickup_longitude BETWEEN -75 AND -73
  AND pickup_latitude BETWEEN 40 AND 42
  AND dropoff_longitude BETWEEN -75 AND -73
  AND dropoff_latitude BETWEEN 40 AND 42
  AND RAND() < 0.001
LIMIT 1000000;



```


> 2nd Command:

```bash

CREATE OR REPLACE TABLE `[YOUR_PROJECT_ID].taxirides.[YOUR_OUTPUT_TABLE]` AS
SELECT
  *
FROM
  ML.PREDICT(
    MODEL `[YOUR_PROJECT_ID].taxirides.[YOUR_MODEL_NAME]`,
    TABLE `[YOUR_PROJECT_ID].taxirides.report_prediction_data`
  );

```
